### 1.4 竞争格局地图存在的代码问题

```
┌─────────────────────────────────────────────────────────────────┐
│                                                                 │
│  ❌ 问题1：气泡大小计算不合理                                    │
│  ──────────────────────────                                     │
│  sizes = self.analysis_df['战略价值'] * 2                        │
│                                                                 │
│  战略价值 = total_score * 0.6 + (100 - abs(change)) * 0.4       │
│  当 total_score ≈ 29, change ≈ 50000+ 时：                      │
│  战略价值 ≈ 29*0.6 + (100 - 50000)*0.4 = 负值！                  │
│  sizes = 负值 * 2 → matplotlib会将其解释为0或极小                 │
│  → 这就是为什么3D图中看不到数据点                                 │
│                                                                 │
│  ❌ 问题2：坐标尺度差异巨大                                      │
│  ──────────────────────────                                     │
│  X轴(排名)：0 ~ 1,000,000                                      │
│  Y轴(变化)：-400,000 ~ 600,000                                  │
│  Z轴(得分)：25 ~ 50                                             │
│  尺度差距达到 20,000倍，3D可视化严重失真                          │
│                                                                 │
│  ❌ 问题3：3D渲染中文字体问题                                    │
│  ──────────────────────────                                     │
│  matplotlib的mpl_toolkits.mplot3d模块对中文支持不佳              │
│  图例文字显示为方块□□                                            │
│                                                                 │
│  ❌ 问题4：战略价值公式在大排名值下失效                            │
│  ──────────────────────────                                     │
│  公式：strategic_value = total_score * 0.6                       │
│                        + (100 - abs(change)) * 0.4               │
│  当排名变化 abs(change) = 50,000+ 时                             │
│  (100 - 50000) * 0.4 = -19,960                                  │
│  → 战略价值为巨大负数 → 所有下游计算失效                          │
│                                                                 │
│  ★ 根本原因：公式设计假设排名变化在0~100范围内，                   │
│    但实际ABA搜索频率排名变化范围在 0~1,000,000                    │
│    → 公式设计与实际数据尺度严重不匹配                              │
│                                                                 │
└─────────────────────────────────────────────────────────────────┘
```

```python
# ━━━━━ 修改前（排名变化7.py 第376-388行）━━━━━
def calculate_rank_score(rank):
    if rank <= 10:      return 100 - (rank - 1) * 5
    elif rank <= 30:    return 70 - (rank - 11) * 2
    elif rank <= 50:    return 40 - (rank - 31) * 1.5
    elif rank <= 100:   return 20 - (rank - 51) * 0.4

# ━━━━━ 修改后 — 适配ABA搜索频率排名(1~2,000,000) ━━━━━
def calculate_rank_score(rank):
    """
    ABA搜索频率排名得分计算
    排名范围：1 ~ 2,000,000+
    得分范围：0 ~ 100
    采用对数尺度映射，避免大排名值导致的得分塌缩
    """
    if rank <= 0:
        return 0
    
    import math
    
    # 使用对数尺度：将 1~2,000,000 映射到 100~0
    # log10(1) = 0 → 得分100
    # log10(100) = 2 → 得分80
    # log10(10,000) = 4 → 得分60
    # log10(100,000) = 5 → 得分50
    # log10(1,000,000) = 6 → 得分20
    # log10(2,000,000) ≈ 6.3 → 得分0
    
    log_rank = math.log10(max(rank, 1))
    
    # 线性映射：log_rank ∈ [0, 6.5] → score ∈ [100, 0]
    score = max(0, 100 - (log_rank / 6.5) * 100)
    
    # 额外奖励：排名特别好的词给加分
    if rank <= 1000:
        score = min(100, score + 10)  # Top 1000 额外+10
    elif rank <= 10000:
        score = min(100, score + 5)   # Top 10000 额外+5
    
    return round(score, 1)
```

```python
# ━━━━━ 修改后 — 基于变异系数(CV)而非绝对标准差 ━━━━━
def calculate_stability_score(rank_values):
    """
    稳定性得分：使用变异系数(CV)代替绝对标准差
    CV = std / mean，无量纲，适用于任何尺度
    """
    if len(rank_values) <= 1:
        return 70
    
    mean_rank = np.mean(rank_values)
    std_rank = np.std(rank_values)
    
    if mean_rank == 0:
        return 50
    
    cv = std_rank / mean_rank  # 变异系数
    
    if cv < 0.05:       # 波动<5%
        return 95
    elif cv < 0.10:     # 波动<10%
        return 85
    elif cv < 0.20:     # 波动<20%
        return 70
    elif cv < 0.35:     # 波动<35%
        return 55
    elif cv < 0.50:     # 波动<50%
        return 40
    else:               # 波动>50%
        return 25
```

### 4.4 战略价值公式修复

```python
# ━━━━━ 修改前（第453行）━━━━━

# ❌ 当 abs(change)=50000 时，(100-50000)*0.4 = -19960

# ━━━━━ 修改后 ━━━━━
def calculate_strategic_value(total_score, change, initial_rank):
    """
    战略价值：综合得分 + 变化趋势加权
    确保输出始终在 0~100 范围
    """
    import math
    
    # 变化率（相对值）
    if initial_rank > 0:
        change_rate = abs(change) / initial_rank
    else:
        change_rate = 0
    
    # 将变化率映射到0~100的稳定性分数
    # change_rate=0 → 100 (完全稳定)
    # change_rate=0.5 → 50
    # change_rate=1.0 → 25
    # change_rate>2.0 → 接近0
    stability_component = max(0, 100 * math.exp(-change_rate * 2))
    
    # 综合战略价值
    strategic_value = total_score * 0.6 + stability_component * 0.4
    
    return round(max(0, min(100, strategic_value)), 1)
```

```python
def calculate_roi_expectation(final_rank, grade_score, total_score):
    """
    ROI预期：基于对数排名和综合评分
    输出范围：-100 ~ 100
    """
    import math
    
    # 排名因子：对数尺度映射到 0~1
    if final_rank > 0:
        rank_factor = max(0, 1 - math.log10(final_rank) / 6.5)
    else:
        rank_factor = 1
    
    # ROI预期 = 排名因子 × 评级因子 × 综合得分因子
    roi = rank_factor * (grade_score + 1) * total_score * 0.02
    
    # 限制范围
    return round(max(-100, min(100, roi)), 2)
```

```python
# ============ 修复版 — 基于真实数据的竞争分析 ============

def _analyze_competitors(self):
    """
    分析竞争对手模式 — 修复版
    
    策略1：如果ABA数据中有Top3 ASIN字段，基于真实数据分析
    策略2：如果没有，基于关键词语义相似度推断竞争关系
    策略3：兜底方案使用确定性计算代替随机数
    """
    if len(self.analysis_df) == 0:
        return
    
    all_keywords = set(self.analysis_df['搜索词'])
    total_kw = len(all_keywords)
    competitor_data = {}
    
    # 检查是否有ASIN相关列
    asin_columns = [col for col in self.df.columns 
                    if 'asin' in col.lower() or 'ASIN' in col]

        # === 策略1：基于ABA Top3 ASIN真实数据 ===
        # 统计出现频率最高的ASIN作为竞争对手
        asin_counter = Counter()
        for col in asin_columns:
            for asin in self.df[col].dropna():
                asin_str = str(asin).strip()
                if len(asin_str) == 10 and asin_str.startswith('B0'):
                    asin_counter[asin_str] += 1
        
        # 取Top3 ASIN作为竞争对手
        top_competitors = asin_counter.most_common(3)
        
        for i, (asin, count) in enumerate(top_competitors):
            comp_name = f'竞争对手{chr(65+i)} ({asin})'
```
